# The checks of the public headers (HeaderChecks; 4.9, 9.1, 9.2, 9.3, 9.7). Added by
# tests/CMakeLists.txt only when AMARK_BUILD_TESTS is on.
#
# amark_header_check (OBJECT library, built with everything else, so a failure is
#   a build failure): for each include/amark/*.h, a C source generated at configure
#   time includes the header, checks that it defined its include guard
#   AMARK_<NAME>_H, and includes it again. It is compiled as C99 with the warnings
#   of amark_target_defaults(), so a header that needs another one it does not
#   include, or that cannot be included twice, fails the build. The list of headers
#   is globbed with CONFIGURE_DEPENDS: adding a header reconfigures.
#
# cxx_include_test (only when a C++ compiler is available, check_language(CXX)):
#   a C++11 Naos test, cxx_include_test.cpp, that includes only <amark/amark.h> and
#   calls a function of every public header that declares one, linked with amark
#   (static or shared as configured), so the functions must link with their C names. The same generated
#   per-header sources are also compiled into it as C++, so that each header
#   compiles on its own as C++ as well. Without a C++ compiler it is not built and
#   a status message says so.
#
# lint.public_headers (label "lint"): cmake/scripts/check_public_headers.cmake
#   checks the prefixes of the public names and that amark.h includes every public
#   header in the canonical include order (design, PublicHeaders).

set(_amark_header_dir "${PROJECT_SOURCE_DIR}/include/amark")
file(GLOB _amark_headers LIST_DIRECTORIES false CONFIGURE_DEPENDS
  RELATIVE "${_amark_header_dir}" "${_amark_header_dir}/*.h")
list(SORT _amark_headers)

# One source per header and language; file(CONFIGURE) rewrites a source only when
# its content changes, so reconfiguring does not rebuild the checks.
set(_amark_header_c_sources "")
set(_amark_header_cxx_sources "")
foreach(_amark_header IN LISTS _amark_headers)
  string(REGEX REPLACE "\\.h$" "" _amark_stem "${_amark_header}")
  string(MAKE_C_IDENTIFIER "${_amark_stem}" _amark_stem)
  string(TOUPPER "AMARK_${_amark_stem}_H" _amark_guard)
  set(_amark_content [=[
/* Generated by tests/headers/CMakeLists.txt: <amark/@_amark_header@> compiles on its
 * own, defines its include guard and can be included twice (4.9). */
#include <amark/@_amark_header@>
#ifndef @_amark_guard@
#error "<amark/@_amark_header@> does not define its include guard @_amark_guard@"
#endif
#include <amark/@_amark_header@>

/* A translation unit must declare something (ISO C). */
typedef int amark_header_check_@_amark_stem@_t;
]=])
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/c/${_amark_stem}_h.c"
    CONTENT "${_amark_content}" @ONLY)
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cxx/${_amark_stem}_h.cpp"
    CONTENT "${_amark_content}" @ONLY)
  list(APPEND _amark_header_c_sources "${CMAKE_CURRENT_BINARY_DIR}/c/${_amark_stem}_h.c")
  list(APPEND _amark_header_cxx_sources
    "${CMAKE_CURRENT_BINARY_DIR}/cxx/${_amark_stem}_h.cpp")
endforeach()

add_library(amark_header_check OBJECT ${_amark_header_c_sources})
amark_target_defaults(amark_header_check)
# The include directories and the definitions (AMARK_SHARED) a user of amark gets.
target_link_libraries(amark_header_check PRIVATE amark)

# The C++ test. check_language() does not look again when CMAKE_CXX_COMPILER is
# already set, so a compiler given on the command line must also exist.
include(CheckLanguage)
check_language(CXX)
set(_amark_cxx_available FALSE)
if(CMAKE_CXX_COMPILER)
  list(GET CMAKE_CXX_COMPILER 0 _amark_cxx_program)
  if(IS_ABSOLUTE "${_amark_cxx_program}")
    if(EXISTS "${_amark_cxx_program}" AND NOT IS_DIRECTORY "${_amark_cxx_program}")
      set(_amark_cxx_available TRUE)
    endif()
  else()
    find_program(_amark_cxx_found NAMES "${_amark_cxx_program}" NO_CACHE)
    if(_amark_cxx_found)
      set(_amark_cxx_available TRUE)
    endif()
    unset(_amark_cxx_found)
  endif()
  unset(_amark_cxx_program)
endif()

if(_amark_cxx_available)
  enable_language(CXX)
  # Not amark_add_test(): amark_target_defaults() sets C options (such as MSVC's
  # /std:c11) that do not apply to C++. The C++ settings are the same in spirit:
  # C++11 without extensions, the same warnings, -Werror or /WX with
  # AMARK_WARNINGS_AS_ERRORS, and the test directory that holds amark's DLL.
  naos_add_executable(cxx_include_test
    SOURCES cxx_include_test.cpp ${_amark_header_cxx_sources}
    LIBRARIES amark)
  set_target_properties(cxx_include_test PROPERTIES
    CXX_STANDARD 11
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(cxx_include_test PRIVATE /W4 /utf-8)
    if(AMARK_WARNINGS_AS_ERRORS)
      target_compile_options(cxx_include_test PRIVATE /WX)
    endif()
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    target_compile_options(cxx_include_test PRIVATE -Wall -Wextra -Wpedantic)
    if(AMARK_WARNINGS_AS_ERRORS)
      target_compile_options(cxx_include_test PRIVATE -Werror)
    endif()
  endif()
else()
  message(STATUS "Amark: no C++ compiler found, cxx_include_test is not built")
endif()
unset(_amark_cxx_available)

add_test(NAME lint.public_headers
  COMMAND "${CMAKE_COMMAND}"
    "-DHEADER_DIR=${_amark_header_dir}"
    -P "${PROJECT_SOURCE_DIR}/cmake/scripts/check_public_headers.cmake")
set_tests_properties(lint.public_headers PROPERTIES LABELS lint)

unset(_amark_header_dir)
unset(_amark_headers)
unset(_amark_header_c_sources)
unset(_amark_header_cxx_sources)
unset(_amark_header)
unset(_amark_stem)
unset(_amark_guard)
unset(_amark_content)
